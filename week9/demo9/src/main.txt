use std::ffi::CString;

/*
Unsafe:




Raw pointers:
- *const T 
- *mut T 

Safe Abstraction:


Soundness:
1. Safety Comments
2. Runtime validation (assert!)
3. Keep minimum unsafe code blocks

FFI:


*/

fn main() {
    println!("=== unsafe ===");
    {
        let mut message = String::from("hello");
        let ptr: *mut String = &mut message;

        unsafe {
            // Here we are directly modifying the value behind a pointer.
            // This is only allowed inside an unsafe block.
            (*ptr).push_str(" world");
            println!("mutated through raw pointer: {}", *ptr);
        }
    }

    println!("\n=== raw pointers ===");
    {
        let numbers = [10, 20, 30, 40];
        let ptr = numbers.as_ptr();
        let mut value = 42;
        let mut_ptr: *mut i32 = &mut value;

        unsafe {
            // *const T: read-only access through the pointer.
            println!("first element: {}", *ptr);
            println!("second element: {}", *ptr.add(1));

            // *mut T: we can also modify the value it points to.
            println!("value before mutation: {}", *mut_ptr);
            *mut_ptr = 100;
            println!("value after mutation: {}", *mut_ptr);
        }
    }

    println!("\n=== safe abstraction ===");
    {
        // This function uses unsafe internally, but the user-facing API is safe.
        // The caller does not need to think about raw pointers.
        fn split_at_mut<T>(slice: &mut [T], mid: usize) -> (&mut [T], &mut [T]) {
            assert!(mid <= slice.len(), "split index out of bounds");

            let len = slice.len();
            let ptr = slice.as_mut_ptr();

            // SAFETY:
            // - mid <= len ensures both resulting slices are within bounds.
            // - The original slice is borrowed mutably, so the two returned slices
            //   are non-overlapping in normal use.
            unsafe {
                (
                    std::slice::from_raw_parts_mut(ptr, mid),
                    std::slice::from_raw_parts_mut(ptr.add(mid), len - mid),
                )
            }
        }

        let mut data = vec![1, 2, 3, 4, 5];
        let (left, right) = split_at_mut(&mut data, 2);

        left[0] = 10;
        right[0] = 20;

        println!("data after split: {:?}", data);
    }

    println!("\n=== FFI ===");
    {
        // FFI lets Rust call functions from C.
        // Since C can do anything with memory, this is unsafe.
        unsafe extern "C" {
            fn puts(s: *const i8) -> i32;
        }

        let message = CString::new("Hello from C").unwrap();
        unsafe {
            let res = puts(message.as_ptr());
        }
    }
}
